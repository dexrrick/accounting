const sha256 = value => createHash('sha256').update(value).digest('hex');
const stableJson = value => JSON.stringify(value);
const absolute = relative => path.resolve(ROOT, relative);
const logPhase = (phase, status, details = {}) => {
  console.log(JSON.stringify({ phase, status, ...details }));
};

async function readRaw(relative) {
  return await readFile(absolute(relative));
}

async function assertRawHash(relative, expected) {
  const actual = sha256(await readRaw(relative));
  assert.equal(actual, expected, `CAPTURE_PREFLIGHT_RAW_HASH_MISMATCH:${relative}`);
  return actual;
}

async function readJson(relative) {
  return JSON.parse((await readRaw(relative)).toString('utf8'));
}

async function assertPathAbsent(relative) {
  try {
    await access(absolute(relative));
    throw new Error(`CAPTURE_OUTPUT_ALREADY_EXISTS:${relative}`);
  } catch (error) {
    if (error?.message?.startsWith('CAPTURE_OUTPUT_ALREADY_EXISTS:')) throw error;
    if (error?.code !== 'ENOENT') throw error;
  }
}

async function writeAuditOnce(document) {
  const destination = absolute(OUTPUTS.result);
  await mkdir(path.dirname(destination), { recursive: true });
  const handle = await open(destination, 'wx');
  const bytes = Buffer.from(JSON.stringify(document, null, 2) + '\n', 'utf8');
  try {
    await handle.writeFile(bytes);
    await handle.sync();
  } finally {
    await handle.close();
  }
  return sha256(bytes);
}

function safeFailure(error) {
  return String(error?.message || error || 'UNKNOWN_CAPTURE_FAILURE').replace(/[\r\n\0]/g, ' ').slice(0, 500);
}

function countHttpStatuses(entries = []) {
  const counts = {};
  for (const entry of entries) {
    const key = entry.failure ? `FAILED_${entry.failure}` : `HTTP_${entry.status}`;
    counts[key] = (counts[key] || 0) + 1;
  }
  return counts;
}

      httpStatusCounts: countHttpStatuses(capturePayload.entries),
      terminalIntegrityFailure: capturePayload.terminalIntegrityFailure,
      evidenceLockCreated: false
    };
  } catch (payloadError) {
    if (payloadError?.code !== 'ENOENT') report.capturePayloadReadFailure = safeFailure(payloadError);
  }
  logPhase(phase, 'FAILED', { code: report.failure.code,
    runnerDispatchCount: report.capture?.runnerDispatchCount ?? null,
    recordedEntryCount: report.capture?.recordedEntryCount ?? null });
}

report.finishedAt = new Date().toISOString();
try {
  report.resultSha256 = await writeAuditOnce(report);
  auditWritten = true;
  logPhase('AUDIT_RESULT', 'SAVED_WITH_EXCLUSIVE_CREATE', { path: OUTPUTS.result, sha256: report.resultSha256 });
} catch (error) {
  logPhase('AUDIT_RESULT', 'SAVE_FAILED', { code: safeFailure(error), priorAuditPreserved: true });
}

if (!auditWritten || report.status !== 'OFFICIAL_CAPTURE_COMPLETE_SEMANTIC_ACCEPTANCE_HELD') {
  process.exitCode = 1;
}
