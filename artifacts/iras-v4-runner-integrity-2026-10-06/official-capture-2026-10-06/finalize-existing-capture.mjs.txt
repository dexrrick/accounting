const abs = p => path.resolve(ROOT, p), hash = b => createHash('sha256').update(b).digest('hex');
const raw = p => readFile(abs(p)), doc = b => JSON.parse(b.toString('utf8'));
async function originals() { for (const [key, expected] of Object.entries(HASH))
  assert.equal(hash(await raw(FILE[key])), expected, `ORIGINAL_${key.toUpperCase()}_CHANGED`); }
async function absent(p) {
  try { await access(abs(p)); } catch (e) { if (e.code === 'ENOENT') return; throw e; }
  throw new Error(`FINALIZATION_OUTPUT_ALREADY_EXISTS:${p}`);
}
async function writeOnce(p, value) {
  const bytes = Buffer.from(`${JSON.stringify(value, null, 2)}\n`), h = await open(abs(p), 'wx');
  try { await h.writeFile(bytes); await h.sync(); } finally { await h.close(); } return hash(bytes);
}
